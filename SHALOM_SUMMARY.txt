     # Opción B: FastAPI (recomendado)
     @router.post("/track")
     async def track(request, service = Depends(get_service)):
         return await service.track_order(request)


💾 INTEGRACIÓN FASTAPI
═══════════════════════════════════════════════════════════════════════════

  # backend/routes/shalom_routes.py
  from fastapi import APIRouter, Depends
  from backend.shalom_service import ShalomService, get_service

  router = APIRouter(prefix="/api/shalom")

  @router.post("/track")
  async def track_order(request, service: ShalomService = Depends(get_service)):
      return await service.track_order(request)

  @router.get("/agencies")
  async def list_agencies(service: ShalomService = Depends(get_service)):
      return await service.list_agencies(per_page=10)

  # Incluir en main.py
  app.include_router(shalom_router)

📊 CAPACIDADES
═══════════════════════════════════════════════════════════════════════════

  RATE LIMITING (Automático):
    • Límite: 60 requests/minuto
    • Si se excede: espera transparente ~60 segundos
    • Continúa automáticamente
    • Logging de eventos

  REINTENTOS:
